cmake_minimum_required(VERSION 3.16)
project(sh2zam VERSION 0.2.0 LANGUAGES C)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(SH2Z_TOP_LEVEL ON)
endif()
option(SH2Z_BUILD_TESTS "Build host correctness tests" ${SH2Z_TOP_LEVEL})
option(SH2Z_BUILD_EXAMPLES "Build executable host examples" ${SH2Z_TOP_LEVEL})
option(SH2Z_FORCE_PORTABLE "Disable SH-2 inline assembly" OFF)
option(SH2Z_SANITIZE "Host AddressSanitizer and UBSan" OFF)
option(SH2Z_STRICT "Treat compiler warnings as errors" OFF)
option(SH2Z_ENABLE_LTO "Opt-in link-time optimization (benchmark first)" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(SH2Z_SANITIZE AND NOT CMAKE_CROSSCOMPILING)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(SH2Z_SOURCES src/sh2zam.c src/sh2zam_trig.c src/sh2zam_angles.c src/sh2zam_transform.c src/sh2zam_geometry.c)
add_library(sh2zam STATIC ${SH2Z_SOURCES})
add_library(SH2ZAM::sh2zam ALIAS sh2zam)
target_include_directories(sh2zam PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(sh2zam PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sh2zam PRIVATE -Wall -Wextra -Wconversion -Wshadow -ffunction-sections -fdata-sections)
  if(SH2Z_STRICT)
    target_compile_options(sh2zam PRIVATE -Werror -Wpedantic)
  endif()
endif()
if(SH2Z_FORCE_PORTABLE)
  target_compile_definitions(sh2zam PRIVATE SH2Z_FORCE_PORTABLE=1)
endif()
if(SH2Z_ENABLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_error LANGUAGES C)
  if(NOT ipo_ok)
    message(FATAL_ERROR "LTO unavailable: ${ipo_error}")
  endif()
  set_property(TARGET sh2zam PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if((SH2Z_BUILD_TESTS OR SH2Z_BUILD_EXAMPLES) AND NOT CMAKE_CROSSCOMPILING)
  add_library(sh2zam_example_kernels STATIC examples/math_examples.c examples/saturn_smoke.c)
  target_link_libraries(sh2zam_example_kernels PUBLIC sh2zam)
  target_include_directories(sh2zam_example_kernels PUBLIC examples)
endif()
if(SH2Z_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_executable(sh2zam_tests tests/test_sh2zam.c)
  target_link_libraries(sh2zam_tests PRIVATE sh2zam)
  add_test(NAME legacy_regressions COMMAND sh2zam_tests)
  add_executable(sh2zam_foundation tests/test_foundation.c)
  target_link_libraries(sh2zam_foundation PRIVATE sh2zam m)
  add_test(NAME foundation_oracles COMMAND sh2zam_foundation)
  enable_language(CXX)
  add_executable(sh2zam_cpp tests/test_cpp.cpp)
  set_target_properties(sh2zam_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  target_link_libraries(sh2zam_cpp PRIVATE sh2zam)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sh2zam_cpp PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)
  endif()
  add_test(NAME cpp11_interop COMMAND sh2zam_cpp)
endif()
if(SH2Z_BUILD_EXAMPLES AND NOT CMAKE_CROSSCOMPILING)
  foreach(example orbit cube batch smoke_main)
    add_executable(sh2zam_${example} examples/${example}.c)
    target_link_libraries(sh2zam_${example} PRIVATE sh2zam_example_kernels)
    if(SH2Z_BUILD_TESTS)
      add_test(NAME example_${example} COMMAND sh2zam_${example})
    endif()
  endforeach()
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS sh2zam EXPORT SH2ZAMTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SH2ZAMTargets NAMESPACE SH2ZAM:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SH2ZAM)
configure_package_config_file(cmake/SH2ZAMConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SH2ZAMConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SH2ZAM)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SH2ZAMConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SH2ZAMConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/SH2ZAMConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SH2ZAM)
