cmake_minimum_required(VERSION 3.13)
project(sh2zam VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SH2Z_BUILD_TESTS "Build portable host-side arithmetic tests" ON)
option(SH2Z_FORCE_PORTABLE "Disable SH-2 assembly paths for reference checks" OFF)
add_library(sh2zam STATIC src/sh2zam.c src/sh2zam_trig.c)
add_library(SH2ZAM::sh2zam ALIAS sh2zam)
target_include_directories(sh2zam PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(SH2Z_FORCE_PORTABLE)
    target_compile_definitions(sh2zam PRIVATE SH2Z_FORCE_PORTABLE=1)
endif()
if(SH2Z_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_executable(sh2zam_tests tests/test_sh2zam.c)
    target_link_libraries(sh2zam_tests PRIVATE sh2zam)
    add_test(NAME sh2zam-host COMMAND sh2zam_tests)
endif()
install(TARGETS sh2zam ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
