cmake_minimum_required(VERSION 3.16)
project(SH2ZAM VERSION 0.2.0 LANGUAGES C)
# Saturn is the default. The inherited Dreamcast build is archived separately.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(SH2Z_TOP_LEVEL ON)
else()
  set(SH2Z_TOP_LEVEL OFF)
endif()
option(SH2Z_BUILD_TESTS "Build host correctness tests" ${SH2Z_TOP_LEVEL})
option(SH2Z_BUILD_EXAMPLES "Build executable host examples" ${SH2Z_TOP_LEVEL})
if(SH2Z_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
endif()
add_subdirectory(saturn)
